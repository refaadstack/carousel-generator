                            return <IconComp className="w-4 h-4 text-amber-500 animate-pulse" style={{ color: getSlideColors(slide).primary }} />;
                          })()}
                        </div>
                      )}
                      {renderSubtitle(slide, getSlideColors(slide), carousel.theme, false, carousel.fontHeading, carousel.fontBody)}

                          slide.headingSizeOverride === 'md' ? 'text-sm' :
                          slide.headingSizeOverride === 'lg' ? 'text-lg' :
                          slide.headingSizeOverride === 'xl' ? 'text-xl' :
                          slide.headingSizeOverride === '2xl' ? 'text-2xl' : 'text-base'

                          textTransform: slide.titleStyleOverride === 'none' ? 'none' : slide.titleStyleOverride || undefined
                        }}

                                </div>
                              )}

                                fontFamily: slide.bodyFontOverride || carousel.fontBody,
                                color: slide.bodyColorOverride || getSlideColors(slide).text,
                                fontSize: slide.bodySizeOverride || undefined
                              }}>
                                {renderFormattedText(item, getSlideColors(slide), carousel.theme)}
                              </p>
                            </div>
                          );
                        })}
                      </div>
                    )}
                  </div>
                )}

                {/* Bottom Image if position is bottom */}
                {slide.showImage && slide.imageUrl && slide.imagePosition === 'bottom' && (
                  <div className="w-full flex justify-center mt-2.5 select-none">
                    {renderSlideImage(slide, 'max-h-[90px]')}
                  </div>
                )}
              </div>

              {/* Right image column if position is right */}
              {slide.showImage && slide.imageUrl && slide.imagePosition === 'right' && (

                </div>

            {slide.ctaText && (
              <div 
                className={`w-full py-2 px-3 text-center text-[10px] font-bold flex items-center justify-center gap-1.5 transition-all border ${
                  carousel.theme === 'neobrutalism'
                    ? 'bg-[#4d96ff] border-2 border-black text-black rounded-none shadow-[3px_3px_0px_#000] font-black uppercase tracking-wider'
                    : 'rounded-lg shadow-sm'
                }`}
                style={carousel.theme !== 'neobrutalism' ? { 
                  backgroundColor: getSlideColors(slide).primary, 
                  color: '#ffffff',
                  borderColor: `${getSlideColors(slide).text}10`
                } : undefined}
              >
                <span>{slide.ctaText}</span>
                <ArrowRight className="w-3 h-3 animate-pulse text-current" />
              </div>
            )}

            {showSlideBottomWatermark ? (
              <div className="flex items-center justify-between border-t border-dashed pt-2.5 opacity-80" style={{ borderColor: `${getSlideColors(slide).text}20` }}>
                <span className="text-[9px] font-mono tracking-wider opacity-60" style={{ color: getSlideColors(slide).text }}>
                  {getWatermarkText(slide)}
                </span>

                {/* Slider Pagination */}
                <div className="flex items-center gap-2">
                  {/* Progress bar */}
                  <div className="w-14 h-1 bg-slate-200/20 rounded-full overflow-hidden">
                    <div

                      style={{ 
                        backgroundColor: getSlideColors(slide).primary,
                        width: `${((idx + 1) / carousel.slides.length) * 100}%` 
                      }}
                    />
                  </div>
                  <span className="text-[10px] font-mono font-bold" style={{ color: getSlideColors(slide).text }}>
                    {String(idx + 1).padStart(2, '0')} / {String(carousel.slides.length).padStart(2, '0')}
                  </span>
                </div>
              </div>
            ) : (
              <div className="flex items-center justify-end pt-1 opacity-80">
                {/* Slider Pagination */}
                <div className="flex items-center gap-2">
                  <div className="w-14 h-1 bg-slate-200/20 rounded-full overflow-hidden">
                    <div 
                      className="h-full rounded-full transition-all duration-300"
                      style={{ 
                        backgroundColor: getSlideColors(slide).primary,
                        width: `${((idx + 1) / carousel.slides.length) * 100}%` 
                      }}
                    />
                  </div>
                  <span className="text-[10px] font-mono font-bold" style={{ color: getSlideColors(slide).text }}>
                    {String(idx + 1).padStart(2, '0')} / {String(carousel.slides.length).padStart(2, '0')}
